C2000 FIFO ISR 提醒

概念版 
interrupt void spi_rx_isr(void)
{
    while (RXFFST >= WORDS_PER_PACKET) {
        read_one_packet_from_SPIRXBUF();
        push_packet_to_ring_buffer();
    }

    if (RXFFOVF == 1) {
        record_overrun_error();
        clear_rx_overflow_flag_in_TRM_order();
    }

    clear_rx_interrupt_flag_in_TRM_order();
}
